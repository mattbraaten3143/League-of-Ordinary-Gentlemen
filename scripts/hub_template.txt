function teamMode(id){
  const rk=D.standings.map(String).indexOf(String(id))+1, t=T[id], six=T[D.standings[5]];
  const gb=Math.max(0,((six.w-t.w)+(t.l-six.l))/2), firsts=D.capital[id].firsts;

  let po=null,ch=null;try{const b=draftBase();if(b&&b[id]){po=b[id].po;ch=b[id].ch}}catch(e){}
  const mode=po!=null?(po>=ODDS_CONTEND?"contender":po>=ODDS_BUBBLE?"playoff bubble":"rebuilding")
    :(rk<=4||(cr<=3&&gb<=2))?"contender":((rk>=9&&firsts>=4)||(gb>1.5&&cr>6))?"rebuilding":"playoff bubble";

  return {rk,gb,firsts,mode,cr,po,ch,ownsFirst,label:mode==="contender"?"contender":mode==="playoff bubble"?"bubble team":"seller"};
}

function tradeStory(c){
  const {name,m,ins,outs,net,pct,dl,lu0,fut,dp,dc,own,others,tot,counter}=c, even=pct<10, N=esc(name);
  const RD={1:"first",2:"second",3:"third",4:"fourth"};
  const nm=x=>{if(x.kind==="pick"){const p=x.row,o=String(p[3]);return o===String(c.tid)?`its own ${p[1]} ${RD[p[2]]||"pick"}`:`${esc(tn(o))}'s ${p[1]} ${RD[p[2]]||"pick"}`}
    return esc(x.label.replace(/ \((QB|RB|WR|TE|K|DEF)\)$/,""))};
  const list=a=>{const n=a.map(nm);return n.length<=1?(n[0]||"nothing"):n.slice(0,-1).join(", ")+" and "+n[n.length-1]};
  const futS=Math.abs(fut)>=0.15*Math.max(1,tot)?Math.sign(fut):0, netS=even?0:Math.sign(net);
  const luWord=c.pts?(Math.abs(dl)<0.5?0:Math.abs(dl)<2.5?1:2):(Math.abs(lu0?dl/lu0:0)<0.01?0:Math.abs(lu0?dl/lu0:0)<0.04?1:2);
  const addsProd=luWord>0&&dl>0, sellsProd=luWord>0&&dl<0;
  const dOwn=own?own.now-own.was:0, ownGood=own&&dOwn<=-0.15, ownBad=own&&dOwn>=0.15, othGood=others.some(o=>o.better);
  const locked=m.po!=null&&m.po>=0.9, dChance=locked?dc:dp, chanceWord=locked?"title chances":"playoff chances", cw=Math.abs(dChance)<0.02?0:Math.abs(dChance)<0.07?1:2;
  const hurtIn=ins.filter(x=>x.kind==="player"&&weeksOut(x.row[0])>=3).map(nm);
  const P1=[],P2=[];
  // what it is
  P1.push(`${N} would be getting ${list(ins)} for ${list(outs)}.`);
  P1.push(even?"Value-wise that's close to a wash.":net>0?(pct<25?"That's a slight win on value.":"That's a clear win on value."):(pct<25?"That's a slight overpay on value.":"That's a big overpay on value."));
  if(hurtIn.length)P1.push(`Sleeper doesn't project ${hurtIn.join(" or ")} to play most of the next month, so don't count on ${hurtIn.length>1?"them":"him"} right away.`);
  P1.push(luWord===0?"This year's lineup stays about where it is.":dl>0?`The weekly lineup gets ${luWord===1?"a bit":"a lot"} stronger${cw===0?`, though it barely moves the ${chanceWord}`:`, and the ${chanceWord} go ${dChance>0?"up":"down"} ${cw===1?"a little":"a lot"}`}.`
    :`The weekly lineup gets ${luWord===1?"a bit":"a lot"} weaker${cw===0?`, though the ${chanceWord} hardly change`:`, and the ${chanceWord} drop ${cw===1?"a little":"noticeably"}`}.`);
  if(own&&m.mode!=="contender"){
    if(addsProd&&!ownBad&&own.was<=1.5)P1.push(`Normally adding points hurts a tank, but ${N} is far enough behind that its own first still projects to go first.`);
    else if(addsProd&&ownBad)P1.push(`The catch: those extra points push its own first-round pick later.`);
    else if(sellsProd&&ownGood)P1.push(`Losing those points also helps its own first-round pick, since a lower Max PF means an earlier pick.`);
  }else if(m.mode!=="contender"&&luWord>0)P1.push(`${N} doesn't own its own 2027 first, so ${sellsProd?"getting worse won't earn it a better pick":"adding points doesn't cost it a pick"}.`);
  others.forEach(o=>P1.push(`Side effect: the ${esc(o.year)} first it owns from ${esc(tn(o.orig))} gets ${o.better?"better":"worse"}, because ${esc(tn(o.orig))} ${o.better?"loses":"adds"} talent here.`));
  // fit, the other path, and the fix
  let tag,tone,fit="",alt="";
  const status=m.po==null?"":m.mode==="contender"?(locked?"has a playoff spot all but locked up":"is in good shape for the playoffs"):m.mode==="playoff bubble"?"is right on the playoff bubble":(m.po>=0.15?"is a long shot for the playoffs":"is realistically out of the playoff race");
  if(m.mode==="rebuilding"){

    alt=addsProd?"If it wanted to make a late run anyway, this would help a little, but the odds argue for selling."
      :sellsProd&&m.po!=null&&m.po>=0.2?"If it would rather take a shot at the playoffs, this works against that.":"";

    else if(luWord===0){tag=netS>0?"Value win, no help now":netS<0?"Costs value, no help now":"Neutral now";tone=netS<0?"warn":"good";fit=netS>0?`${N} ${status}, and this adds value without costing anything this year.`:netS<0?`${N} ${status}, and this gives up value without helping this year.`:""}

    else{tag=netS>0?"Value win, small hit now":"Small hit now";tone="warn";fit=netS>0?`${N} ${status}. It gives up a little this year for a clear gain in value.`:`${N} ${status}. It's a small step back this year for little in return.`}
    if(m.mode==="contender")alt=futS>0?"If it ever wants to retool, this helps there too.":futS<0&&pct>=10?"If it pivots to the future later, this is the kind of deal it would wish it had back.":"";
    else alt=sellsProd?((ownGood||futS>0)?`If it decides to sell instead, this is a good first step${ownGood?", since it also improves its own pick":""}.`:"If it decides to sell instead, this doesn't bring back enough for the future.")
      :addsProd?(own?"If it ends up selling instead, this is the wrong move, because the added points push its own pick later.":"If it ends up selling instead, this is the wrong move."):"";
  }
  if(counter&&counter.none)P2.push(`${esc(tn(c.otherId))} doesn't have the picks or young players to make this fair, so ${N} should shop ${list(outs)} elsewhere.`);

}

function tfBase(id){if(TF.lu[id]==null)TF.lu[id]=bestLineup(((D.assets||{})[id]||{players:[]}).players);return TF.lu[id]}
/* Positional need: each team's starters at QB, RB, WR and TE (rest-of-season value) ranked against the league */
const TF_NEED={QB:1,RB:2,WR:3,TE:1};
function tfPos(pl){const o={};for(const pos in TF_NEED)o[pos]=pl.filter(p=>p[2]===pos&&!p[10]).map(p=>p[7]||0).sort((a,b)=>b-a).slice(0,TF_NEED[pos]).reduce((a,b)=>a+b,0);return o}
function tfPosRank(id){if(!TF.pr){TF.pr={};const ids=Object.keys(D.assets||{}),pv={};ids.forEach(i=>pv[i]=tfPos(D.assets[i].players));
  ids.forEach(i=>{TF.pr[i]={};for(const pos in TF_NEED)TF.pr[i][pos]={r:1+ids.filter(j=>pv[j][pos]>pv[i][pos]).length,v:pv[i][pos]}})}return TF.pr[String(id)]}
function tfSide(tid,outs,ins){
  const m=teamMode(tid), pl=((D.assets||{})[tid]||{players:[]}).players, sum=a=>a.reduce((t,x)=>t+x.value,0);
  const gone=new Set(outs.filter(x=>x.kind==="player").map(x=>x.pid));
  const before=tfBase(tid), after=bestLineup([...pl.filter(p=>!gone.has(p[0])),...ins.filter(x=>x.kind==="player").map(x=>x.row)]);
  const [vin,vout]=vmPackage([ins.map(x=>x.value),outs.map(x=>x.value)],VALS.params||{}), net=vin-vout, tot=(vin+vout)||1, pct=Math.abs(net)/tot*100, dl=after-before;
  const young=x=>x.kind==="pick"||(x.age&&x.age<=24), fut=sum(ins.filter(young))-sum(outs.filter(young));
  // need: gains at a weak position count up to 2x, gains at a strong one as little as 0.5x
  const pr=tfPosRank(tid), pa=tfPos([...pl.filter(p=>!gone.has(p[0])),...ins.filter(x=>x.kind==="player").map(x=>x.row)]);
  let need=0;const fills=[],hurts=[];
  for(const pos in TF_NEED){const d=pa[pos]-pr[pos].v,w=0.5+1.5*(pr[pos].r-1)/11;need+=w*d;
    if(pr[pos].r>=8&&d>0.05*Math.max(pr[pos].v,1))fills.push(pos);
    if(pr[pos].r>=8&&d<-0.05*Math.max(pr[pos].v,1))hurts.push(pos)}
  let tag,tone;
  if(m.mode!=="rebuilding"){
    if(dl>0&&(net>=0||pct<=15)){tag="Good for a playoff push";tone="good"}
    else if(dl>=0&&net>0){tag="Value win, lineup intact";tone="good"}
    else if(dl>0){tag="Pricey push";tone="warn"}
    else if(dl===0&&pct<10){tag="Neutral now";tone="warn"}
    else if(net>0){tag="Value win, weaker now";tone="warn"}
    else{tag="Bad for a contender";tone="bad"}
  }else{
    if(net>=0&&fut>0){tag="Good for the rebuild";tone="good"}
    else if(fut>0&&pct<=10){tag="Fair rebuild move";tone="good"}
    else if(fut===0&&net>=0){tag="Neutral for a rebuild";tone="warn"}
    else if(net>0){tag="Value win, older roster";tone="warn"}
    else if(fut>0){tag="Younger, but overpaid";tone="warn"}
    else{tag="Bad for the rebuild";tone="bad"}
  }
  if(m.mode!=="rebuilding"&&fills.length&&tone!=="bad"&&dl>=0){tag=`Fills a need at ${fills.join(" and ")}`;tone="good"}
  return {tid,mode:m.mode,net,pct,dl,fut,tag,tone,before,after,need,fills,hurts};
}

function model(){
  const ids=Object.keys(T), games=D.weeks.reduce((n,w)=>n+1,0);
  const scores={};ids.forEach(id=>scores[id]=[]);
  D.weeks.forEach(w=>w.games.forEach(g=>g.forEach(x=>scores[x.id].push(x.pts))));
  const all=ids.flatMap(id=>scores[id]), mean=all.reduce((a,b)=>a+b,0)/all.length;
  const own={};ids.forEach(id=>{const s=scores[id],n=s.length,avg=s.reduce((a,b)=>a+b,0)/(n||1);own[id]=(n*avg+3*mean)/(n+3)});
  let resid=[];ids.forEach(id=>{const s=scores[id],avg=s.reduce((a,b)=>a+b,0)/(s.length||1);s.forEach(v=>resid.push(v-avg))});
  const sd=Math.max(15,Math.sqrt(resid.reduce((a,b)=>a+b*b,0)/Math.max(1,resid.length-ids.length)));
  let mu={...own}, slope=0;
  if(PR&&PR.teams){
    const sv=ids.map(id=>PR.teams[id].starterValue), sm=sv.reduce((a,b)=>a+b,0)/sv.length;
    const pp=ids.map(id=>scores[id].reduce((a,b)=>a+b,0)/(scores[id].length||1)), pm=pp.reduce((a,b)=>a+b,0)/pp.length;
    const cov=sv.reduce((a,v,i)=>a+(v-sm)*(pp[i]-pm),0), vr=sv.reduce((a,v)=>a+(v-sm)**2,0);
    slope=Math.max(0,vr?cov/vr:0);
    ids.forEach((id,i)=>mu[id]=0.5*own[id]+0.5*(mean+slope*(sv[i]-sm)));
  }
  // Preferred: Sleeper projections of each current roster's best legal lineup, week by week (byes and IR count zero),
  // calibrated to the league's actual scoring level, with 15% weight on scoring so far
  const PL=SIM_PL||D.projLineups, PW=D.projSeason&&D.projSeason.weeks;
  let byWeek=null, src="history", k=1;
  if(PL&&PW&&ids.every(id=>PL[id])){
    const flat=ids.flatMap(id=>PL[id]).filter(v=>v>0); k=flat.length?mean/(flat.reduce((a,b)=>a+b,0)/flat.length):1;
    byWeek={};ids.forEach(id=>{byWeek[id]={};PW.forEach((w,j)=>byWeek[id][w]=0.85*k*PL[id][j]+0.15*own[id])});
    const rs=new Set(SCHED.map(x=>x.week));
    ids.forEach(id=>{const v=PW.filter(w=>rs.has(w)).map(w=>byWeek[id][w]);if(v.length)mu[id]=v.reduce((a,b)=>a+b,0)/v.length});
    src="projections";
  }
  const muAt=(id,w)=>byWeek&&w!=null&&byWeek[id][w]!=null?byWeek[id][w]:mu[id];
  return {ids,mu,sd,mean,games,muAt,src,k};
}
function simulate(N=10000,seed=Date.now()){
  const t0=performance.now(), M=model(), r=rng(seed), ids=M.ids, n=ids.length, P=PO.teams||6;
  const idx={};ids.forEach((id,i)=>idx[id]=i);
  const baseW=ids.map(id=>T[id].w), basePF=ids.map(id=>T[id].pf), baseMX=ids.map(id=>T[id].maxpf||T[id].pf), mu=ids.map(id=>M.mu[id]);
  const gp=ids.map(id=>Math.max(1,T[id].w+T[id].l+T[id].t)), gap=ids.map((id,i)=>(baseMX[i]-basePF[i])/gp[i]);
  const RU=(D.rules&&D.rules.payouts)||[], pay=k=>(RU.find(x=>x.key===k)||{}).amount||0;
  const norm=()=>{let u=0,v=0;while(!u)u=r();while(!v)v=r();return Math.sqrt(-2*Math.log(u))*Math.cos(2*Math.PI*v)};
  const score=(i,w)=>M.muAt(ids[i],w)+M.sd*norm();
  const out={po:new Float64Array(n),bye:new Float64Array(n),one:new Float64Array(n),fin:new Float64Array(n),champ:new Float64Array(n),last:new Float64Array(n),wins:new Float64Array(n),
    pfLead:new Float64Array(n),second:new Float64Array(n),money:new Float64Array(n),seed:ids.map(()=>new Float64Array(n)),slot:ids.map(()=>new Float64Array(n))};
  const wk=new Float64Array(n), pf=new Float64Array(n), mx=new Float64Array(n), order=[...Array(n).keys()];
  for(let k=0;k<N;k++){
    for(let i=0;i<n;i++){wk[i]=baseW[i];pf[i]=basePF[i];mx[i]=baseMX[i]}
    for(let w=0;w<SCHED.length;w++){
      for(const g of SCHED[w].games){
        const a=idx[g[0]],b=idx[g[1]],wn=SCHED[w].week;let sa=score(a,wn),sb=score(b,wn);
        const lock=w===0?simLocks[g.join("-")]:0;
        if(lock===1&&sa<sb)[sa,sb]=[sb,sa]; if(lock===2&&sb<sa)[sa,sb]=[sb,sa];
        pf[a]+=sa;pf[b]+=sb;mx[a]+=sa+gap[a];mx[b]+=sb+gap[b]; if(sa>sb)wk[a]++;else wk[b]++;
      }
    }
    order.sort((x,y)=>wk[y]-wk[x]||pf[y]-pf[x]);
    for(let s=0;s<n;s++){const i=order[s];out.seed[i][s]++;out.wins[i]+=wk[i]-baseW[i]}
    out.one[order[0]]++; out.last[order[n-1]]++;
    for(let s=0;s<P;s++)out.po[order[s]]++;
    out.bye[order[0]]++;out.bye[order[1]]++;
    // bracket: 3v6, 4v5; 1 plays the lowest seed left, 2 the other; QF losers play for 5th, SF losers for 3rd
    const W1=PO.start, W2=PO.start+1, W3=PO.start+2;
    const play=(x,y,w)=>score(x,w)>=score(y,w)?[x,y]:[y,x], sd=s=>order.indexOf(s);
    const [q1w,q1l]=play(order[2],order[5],W1), [q2w,q2l]=play(order[3],order[4],W1), [f5w,f5l]=play(q1l,q2l,W2);
    const [lo,hi]=sd(q1w)>sd(q2w)?[q1w,q2w]:[q2w,q1w];
    const [f1,s1l]=play(order[0],lo,W2), [f2,s2l]=play(order[1],hi,W2), [t3w,t3l]=play(s1l,s2l,W3), [ch,ru]=play(f1,f2,W3);
    out.fin[f1]++;out.fin[f2]++;out.champ[ch]++;out.second[ru]++;
    let ld=0;for(let i=1;i<n;i++)if(pf[i]>pf[ld])ld=i;
    out.pfLead[ld]++;
    out.money[ld]+=pay("pf");out.money[ru]+=pay("second");out.money[ch]+=pay("champ");
    // rookie draft order (league rules): non-playoff teams by reverse Max PF, then 5th W/L, 3rd W/L, runner-up, champion;
    // a points leader who misses the final drops to 1.10

  }

  return {N,ids,out,M,ms};
}

  <section class="rsec" id="hw-values"><h2>How OGSN values work</h2>
    <p>Every value on this site starts from <a href="https://fantasycalc.com" target="_blank" rel="noopener">FantasyCalc</a>'s dynasty trade values for a 1QB, 12-team, full-PPR league. The league's value model then adjusts them:</p>

      <li><b>Position weights.</b> QB ×${B.pos.QB}, RB ×${B.pos.RB}, WR ×${B.pos.WR}, TE ×${B.pos.TE}. Quarterbacks matter less in a 1QB league, and receivers, who hold value longer, are weighted above running backs.</li>
      <li><b>Age curve,</b> on top of what FantasyCalc already prices in (table below).</li>
      <li><b>Draft picks.</b> 2027 picks are valued by their exact projected slot under the <a href="#rules">league's draft-order rules</a>, so a projected 1.01 is worth far more than a late first. 2028 and 2029 picks start from the value of the same tier in 2027 and are discounted for time (×${(B.yearDiscount||{})["2"]} for 2028, ×${(B.yearDiscount||{})["3"]} for 2029), so next year's pick is always worth more than a later one at the same slot.   <section class="rsec" id="hw-trades"><h2>How trades are analyzed</h2>
    <p>Every trade gets the same treatment from code before anything is written about it:</p>
    <ul class="rlist">
      <li><b>Value.</b> Each side's OGSN value received, with the package adjustment. The side that gets at least 10% more of the deal's total value wins; anything closer is called even.</li>
      <li><b>Team situation.</b> Each team's record, standings place and games back of 6th at the time of the trade, its first-round picks and its roster value rank. From these, each team is labeled a <i>contender</i> (top four in the standings), <i>rebuilding</i> (9th or lower with at least four firsts, or more than 1.5 games back of 6th) or <i>playoff bubble</i> (everyone else). The trade calculator goes further and uses the playoff simulator instead (below).</li>
      <li><b>Production.</b> Points per game coming in and going out, and each player's age.</li>
      <li><b>Draft-order effect.</b> The 2027 draft projection is re-run with and without the trade, and the change in value of each side's 2027 picks is added. Selling production lowers a team's Max PF, which can move its own pick up.</li>
    </ul>
    <p>The <a href="#calc">trade calculator</a> uses the same values live and judges each side by its situation in the playoff simulator: 65% playoff odds or better is a contender, 35% or better is on the bubble, and below that the team should be selling. It measures this year's lineup in projected points per week from Sleeper projections, with bench depth filling any spot a trade opens and injured weeks counting as zero, and re-runs the simulation with both rosters changed to see how playoff odds, title odds and the 2027 draft order move. Each side gets a verdict for its situation, a note on how the deal looks if it took the other path (pushing or selling), and, when it loses on value, the smallest addition that would make it fair. Signed-in claude.ai viewers can also ask Claude for a conversational take on the same facts; Claude sees only that trade's facts and verdicts and cannot change the verdict. The <a href="#finder">trade finder</a> searches for one- and two-piece packages close in value that fit both teams' plans. A team trying to win is never asked to give up a starter at one of its weakest positions, and ideas that fill a team's weak spots rank higher. On the <a href="#trades">trade analysis</a> page, Claude writes the headline and each side's take from these computed facts.</p></section>
  <section class="rsec" id="hw-odds"><h2>How the playoff odds work</h2>
    <p>The <a href="#playoffs">playoff odds</a> play out the rest of the season 10,000 times. Each week, every team's score is a random draw centered on Sleeper's projection for the best lineup its current roster can field. That center is scaled to the league's real scoring level, and 15% of the weight stays on each team's scoring so far. The spread matches the league's actual week-to-week swings. Each simulated season runs the remaining schedule, seeds the top six, plays the bracket in Weeks 15 to 17 and pays out the prizes under the <a href="#rules">league rules</a>. The same simulations set the 2027 draft order: non-playoff teams by reverse Max PF, then the consolation and title games. Picking a winner for any of next week's games re-runs everything with that result fixed. Odds change only when scores, rosters or projections change.</p></section>
  <section class="rsec" id="hw-power"><h2>How the power rankings work</h2>
    <p>The <a href="#rankings">power rankings</a> rank every team two ways. <b>Trade value</b> adds up every rostered player's Dynasty Daddy dynasty value plus OGSN values for its draft picks. <b>Contender value</b> is the rest-of-season value of the team's best legal lineup, leaving out injured reserve. The tier comes from both: top four in contender value and top six in trade value is <i>Contender, built to last</i>; top four in contender value alone is <i>Win-now contender</i>; 5th to 8th is <i>In the mix</i>; anyone else in the top six for trade value is <i>Rebuild with assets</i>, and the rest are a <i>Long rebuild</i>.</p></section>
